DROP DATABASE IF EXISTS flashmart_db;
CREATE DATABASE flashmart_db
    CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE flashmart_db;

-- ============================================================
-- 1. TẠO BẢNG
-- ============================================================
CREATE TABLE Customers (
    customer_id INT PRIMARY KEY,
    name        VARCHAR(50) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE Products (
    product_id   INT PRIMARY KEY,
    product_name VARCHAR(50) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE Orders (
    order_id    INT PRIMARY KEY,
    customer_id INT,
    product_id  INT,
    CONSTRAINT fk_orders_customer FOREIGN KEY (customer_id)
        REFERENCES Customers(customer_id),
    CONSTRAINT fk_orders_product FOREIGN KEY (product_id)
        REFERENCES Products(product_id)
) ENGINE=InnoDB;

-- ============================================================
-- ✅ BÁO CÁO 1: MARKETING — ĐẦY ĐỦ KHÁCH HÀNG
-- Dùng LEFT JOIN để giữ toàn bộ khách hàng
-- Dùng COUNT(o.order_id) để khách chưa mua ra 0
-- ============================================================
SELECT
    c.customer_id,
    c.name,
    COUNT(o.order_id) AS total_orders
FROM Customers c
LEFT JOIN Orders o ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.name
ORDER BY c.customer_id;

-- Kết quả kỳ vọng:
-- 1 | Alice   | 2
-- 2 | Bob     | 1
-- 3 | Charlie | 0   ← PHẢI XUẤT HIỆN


-- ============================================================
-- ✅ BÁO CÁO 2: KHO VẬN — SẢN PHẨM CHƯA TỪNG BÁN (ANTI-JOIN)
-- Dùng LEFT JOIN + WHERE IS NULL
-- ============================================================
SELECT
    p.product_id,
    p.product_name
FROM Products p
LEFT JOIN Orders o ON p.product_id = o.product_id
WHERE o.order_id IS NULL
ORDER BY p.product_id;

-- Kết quả kỳ vọng:
-- 103 | Keyboard


-- ============================================================
-- 4. SO SÁNH CÁC CÁCH VIẾT KHÁC (mở rộng)
-- ============================================================

-- 4.1. Anti-Join bằng NOT EXISTS (tối ưu hơn khi bảng Orders lớn)
SELECT p.product_id, p.product_name
FROM Products p
WHERE NOT EXISTS (
    SELECT 1 FROM Orders o WHERE o.product_id = p.product_id
);

-- 4.2. Anti-Join bằng NOT IN (CẢNH BÁO: chậm + sai nếu có NULL)
SELECT p.product_id, p.product_name
FROM Products p
WHERE p.product_id NOT IN (
    SELECT o.product_id FROM Orders o WHERE o.product_id IS NOT NULL
);

-- 4.3. Giả lập FULL OUTER JOIN bằng UNION của LEFT + RIGHT
SELECT c.customer_id, c.name, o.order_id
FROM Customers c LEFT JOIN Orders o ON c.customer_id = o.customer_id
UNION
SELECT c.customer_id, c.name, o.order_id
FROM Customers c RIGHT JOIN Orders o ON c.customer_id = o.customer_id;
