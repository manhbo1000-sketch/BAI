package com.codegym;

import java.security.NoSuchAlgorithmException;

/**
 * Calculator — Thư viện tiện ích toán học.
 * Các phương thức đều là static, có thể gọi trực tiếp không cần khởi tạo.
 */
public class Calculator {

    /**
     * Cộng 2 số nguyên.
     */
    public static int sum(int a, int b) {
        return a + b;
    }

    /**
     * Trừ 2 số nguyên.
     */
    public static int sub(int a, int b) {
        return a - b;
    }

    /**
     * Nhân 2 số nguyên.
     */
    public static int mul(int a, int b) {
        return a * b;
    }

    /**
     * Chia 2 số nguyên.
     * @throws NoSuchAlgorithmException nếu b = 0
     */
    public static int divide(int a, int b) throws NoSuchAlgorithmException {
        if (b == 0) {
            throw new NoSuchAlgorithmException();
        }
        return a / b;
    }
}
